# Omni-Verse Blog — Phase 3 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add authentication (Auth.js v5, GitHub + Google), a libsql/Turso database, and three interactive features — likes, bookmarks (with anonymous→signed-in merge), and flat comments with a markdown subset.

**Architecture:** Drizzle ORM over libsql, DB-backed Auth.js sessions, server actions for all mutations (one route handler kept for `/api/auth/[...nextauth]`). Anonymous users still read everything; sign-in unlocks likes, comments, and DB-synced bookmarks. Phase 1's MDX content stays in files; only user-generated content is in the DB.

**Tech Stack:** `next-auth@5.x`, `@auth/drizzle-adapter`, `drizzle-orm`, `@libsql/client`, `drizzle-kit`. Tests use an in-memory libsql DB (`createClient({ url: ':memory:' })`) with the same schema, fed via the generated migration.

**Source spec:** `docs/superpowers/specs/2026-04-26-omni-verse-blog-phase-3-design.md`
**Phase 2 baseline:** `ea2d439` on `main`.

```ts
import { sqliteTable, text, integer, primaryKey, index } from 'drizzle-orm/sqlite-core';
import { relations, sql } from 'drizzle-orm';

export const users = sqliteTable('user', {
  id: text('id').primaryKey().$defaultFn(() => crypto.randomUUID()),
  name: text('name'),
  email: text('email').unique(),
  emailVerified: integer('emailVerified', { mode: 'timestamp_ms' }),
  image: text('image'),
});

export const accounts = sqliteTable(
  'account',
  {
    userId: text('userId')
      .notNull()
      .references(() => users.id, { onDelete: 'cascade' }),
    type: text('type').notNull(),
    provider: text('provider').notNull(),
    providerAccountId: text('providerAccountId').notNull(),
    refresh_token: text('refresh_token'),
    access_token: text('access_token'),
    expires_at: integer('expires_at'),
    token_type: text('token_type'),
    scope: text('scope'),
    id_token: text('id_token'),
    session_state: text('session_state'),
  },
  (t) => ({ pk: primaryKey({ columns: [t.provider, t.providerAccountId] }) }),
);

export const sessions = sqliteTable('session', {
  sessionToken: text('sessionToken').primaryKey(),
  userId: text('userId')
    .notNull()
    .references(() => users.id, { onDelete: 'cascade' }),
  expires: integer('expires', { mode: 'timestamp_ms' }).notNull(),
});

export const verificationTokens = sqliteTable(
  'verificationToken',
  {
    identifier: text('identifier').notNull(),
    token: text('token').notNull(),
    expires: integer('expires', { mode: 'timestamp_ms' }).notNull(),
  },
  (t) => ({ pk: primaryKey({ columns: [t.identifier, t.token] }) }),
);

export const likes = sqliteTable(
  'likes',
  {
    userId: text('user_id')
      .notNull()
      .references(() => users.id, { onDelete: 'cascade' }),
    postSlug: text('post_slug').notNull(),
    createdAt: integer('created_at', { mode: 'timestamp_ms' })
      .notNull()
      .default(sql`(unixepoch() * 1000)`),
  },
  (t) => ({ pk: primaryKey({ columns: [t.userId, t.postSlug] }) }),
);

export const bookmarks = sqliteTable(
  'bookmarks',
  {
    userId: text('user_id')
      .notNull()
      .references(() => users.id, { onDelete: 'cascade' }),
    postSlug: text('post_slug').notNull(),
    createdAt: integer('created_at', { mode: 'timestamp_ms' })
      .notNull()
      .default(sql`(unixepoch() * 1000)`),
  },
  (t) => ({ pk: primaryKey({ columns: [t.userId, t.postSlug] }) }),
);

export const comments = sqliteTable(
  'comments',
  {
    id: text('id').primaryKey().$defaultFn(() => crypto.randomUUID()),
    userId: text('user_id')
      .notNull()
      .references(() => users.id, { onDelete: 'cascade' }),
    postSlug: text('post_slug').notNull(),
    body: text('body').notNull(),
    createdAt: integer('created_at', { mode: 'timestamp_ms' })
      .notNull()
      .default(sql`(unixepoch() * 1000)`),
    deletedAt: integer('deleted_at', { mode: 'timestamp_ms' }),
  },
  (t) => ({ bySlug: index('comments_by_slug_idx').on(t.postSlug, t.createdAt) }),
);

export const commentsRelations = relations(comments, ({ one }) => ({
  user: one(users, { fields: [comments.userId], references: [users.id] }),
}));
```

```ts
import { createClient } from '@libsql/client';
import { drizzle } from 'drizzle-orm/libsql';
import { migrate } from 'drizzle-orm/libsql/migrator';
import path from 'node:path';
import * as schema from './schema';

export type TestDb = ReturnType<typeof drizzle<typeof schema>>;

export async function makeTestDb(): Promise<TestDb> {
  const client = createClient({ url: ':memory:' });
  const db = drizzle(client, { schema });
  await migrate(db, { migrationsFolder: path.join(process.cwd(), 'drizzle') });

    image: overrides.image ?? 'https://example.test/avatar.png',
```

```tsx
// src/components/UserMenu.tsx
'use client';

import { useEffect, useRef, useState } from 'react';
import Link from 'next/link';
import { signOut } from 'next-auth/react';

type UserLite = { id: string; name?: string | null; image?: string | null };

export function UserMenu({ user }: { user: UserLite }) {
  const [open, setOpen] = useState(false);
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    function onClickOutside(e: MouseEvent) {
      if (!ref.current?.contains(e.target as Node)) setOpen(false);
    }
    function onEsc(e: KeyboardEvent) {
      if (e.key === 'Escape') setOpen(false);
    }
    document.addEventListener('mousedown', onClickOutside);
    document.addEventListener('keydown', onEsc);
    return () => {
      document.removeEventListener('mousedown', onClickOutside);
      document.removeEventListener('keydown', onEsc);
    };
  }, []);

  const initial = (user.name?.[0] ?? 'R').toUpperCase();
  return (
    <div className="user-menu" ref={ref}>
      <button
        type="button"
        className="user-menu-button"
        aria-haspopup="menu"
        aria-expanded={open}
        onClick={() => setOpen((v) => !v)}
      >
        {user.image ? (
          // eslint-disable-next-line @next/next/no-img-element
          <img src={user.image} alt="" className="user-avatar" />
        ) : (
          <span className="user-avatar user-avatar-fallback" aria-hidden="true">
            {initial}
          </span>
        )}
      </button>
      {open && (
        <div className="user-menu-panel" role="menu">
          <Link href="/bookmarks" className="user-menu-item" role="menuitem">
            Bookmarks
          </Link>
          <button
            type="button"
            className="user-menu-item"
            role="menuitem"
            onClick={() => signOut({ redirectTo: '/' })}
          >
            Sign out
          </button>
        </div>
      )}
    </div>
  );
}
```

```ts
const ENTITIES: Record<string, string> = {
  '&': '&amp;',
  '<': '&lt;',
  '>': '&gt;',
  '"': '&quot;',
  "'": '&#39;',
};

function escapeHtml(s: string): string {
  return s.replace(/[&<>"']/g, (c) => ENTITIES[c]!);
}

const ALLOWED_SCHEME = /^(?:https?:|mailto:)/i;

/**
 * Render a tiny markdown subset to a safe HTML string.
 * Supports: **bold**, *italic*, `code`, [text](url), \n -> <br />.
 * Everything else is HTML-escaped. Unknown link schemes render as the literal source.
 */
export function renderMarkdownSubset(input: string): string {
  if (!input) return '';

  // Pass 1: extract code spans (their content is escaped but not further parsed).
  // Replace each `…` with a placeholder so subsequent passes don't see its content.
  const codes: string[] = [];
  let text = input.replace(/`([^`\n]+)`/g, (_m, body: string) => {
    codes.push(`<code>${escapeHtml(body)}</code>`);
    return ` C${codes.length - 1} `;
  });

  // Pass 2: extract links (rejecting disallowed schemes).
  const links: string[] = [];
  text = text.replace(/\[([^\]]+)\]\(([^)\s]+)\)/g, (m, label: string, url: string) => {
    if (!ALLOWED_SCHEME.test(url)) return m; // render as literal
    links.push(
      `<a href="${escapeHtml(url)}" target="_blank" rel="noreferrer">${escapeHtml(label)}</a>`,
    );
    return ` L${links.length - 1} `;
  });

  // Pass 3: escape everything else.
  text = escapeHtml(text);

  // Pass 4: bold, then italic. Bold first so `**foo *bar***` becomes <strong>foo <em>bar</em></strong>.
  text = text.replace(/\*\*([^*\n]+?)\*\*/g, '<strong>$1</strong>');
  text = text.replace(/\*([^*\n]+?)\*/g, '<em>$1</em>');

  // Pass 5: newlines.
  text = text.replace(/\n/g, '<br />');

  // Pass 6: re-insert code and link placeholders. Re-escape the placeholder markers
  // (they were escaped in pass 3 — but   isn't a special HTML char, so escapeHtml
  // leaves it alone; we just need to find the literal markers). The escape pass also
  // escaped any & in our pre-built HTML strings (& → &amp;), but those strings only
  // appeared after pass-3 input, so they're untouched.
  text = text.replace(/ C(\d+) /g, (_m, i: string) => codes[Number(i)]!);
  text = text.replace(/ L(\d+) /g, (_m, i: string) => links[Number(i)]!);

  return text;
}
```

```ts
const lastPost = new Map<string, number>();
const WINDOW_MS = 30_000;

export type Verdict =
  | { allowed: true }
  | { allowed: false; waitMs: number };

export function check(userId: string): Verdict {
  const now = Date.now();
  const last = lastPost.get(userId);
  if (last !== undefined && now - last < WINDOW_MS) {
    return { allowed: false, waitMs: WINDOW_MS - (now - last) };
  }
  lastPost.set(userId, now);
  return { allowed: true };
}

/** Test-only helper. */
export function reset(): void {
  lastPost.clear();
}
```

```ts
  const existing = await db
    .select()
    .from(likes)
    .where(and(eq(likes.userId, userId), eq(likes.postSlug, slug)));

      .delete(likes)

    .select({ c: count() })
    .from(likes)
    .where(eq(likes.postSlug, slug));
```

```ts
export async function mergeBookmarksImpl(

  localSlugs: string[],

  if (localSlugs.length === 0) return { added: 0 };
  const rows = localSlugs.map((slug) => ({ userId, postSlug: slug }));

  slug: string,
```

```tsx
  if (typeof window === 'undefined') return [];
  try {
    const raw = window.localStorage.getItem(STORAGE_KEY);
    return raw ? (JSON.parse(raw) as string[]) : [];
  } catch {
    return [];
  }
}

function writeLocal(slugs: string[]): void {
  try {
    window.localStorage.setItem(STORAGE_KEY, JSON.stringify(slugs));
  } catch {
    // ignore — private mode etc.
  }
}

export function BookmarkButton({
  slug,
  initialBookmarked,
}: {
  slug: string;
  initialBookmarked?: boolean;
}) {
  const session = useSession();
  const router = useRouter();
  const pathname = usePathname() ?? `/posts/${slug}`;
  const toast = useToast();
  const [bookmarked, setBookmarked] = useState<boolean>(initialBookmarked ?? false);
  const [pending, startTransition] = useTransition();

  // Hydrate from localStorage on mount when anonymous.
  useEffect(() => {
    if (session.status !== 'authenticated' && initialBookmarked === undefined) {
      setBookmarked(readLocal().includes(slug));
    }
  }, [session.status, initialBookmarked, slug]);

  function onClick() {
    if (session.status !== 'authenticated') {
      // Anonymous flow — localStorage only.
      const slugs = readLocal();
      const has = slugs.includes(slug);
      const next = has ? slugs.filter((s) => s !== slug) : [...slugs, slug];
      writeLocal(next);
      setBookmarked(!has);
      toast.show(has ? 'Removed bookmark' : 'Bookmarked');
      return;
    }
    // Signed-in flow — server action.
    startTransition(async () => {
      const prev = bookmarked;
      setBookmarked(!prev);
      try {
        const result = await toggleBookmark(slug);
        setBookmarked(result.bookmarked);
        toast.show(result.bookmarked ? 'Bookmarked' : 'Removed bookmark');
      } catch {
        setBookmarked(prev);
        toast.show('Could not update bookmark');
      }
    });
  }

  void router; // reserved for "sign in to bookmark on a future protected page"
  void pathname;

  return (
    <button
      type="button"
      className="bookmark-button"
      data-bookmarked={bookmarked ? 'true' : 'false'}
      aria-pressed={bookmarked}
      aria-label={bookmarked ? 'Remove bookmark' : 'Bookmark this post'}
      disabled={pending}
      onClick={onClick}
    >
      <svg
        width="14"
        height="14"
```

```tsx
'use client';

import { useEffect, useRef } from 'react';
import { useSession } from 'next-auth/react';
import { mergeBookmarks } from '@/lib/bookmarks';
import { useToast } from './ToastProvider';

const STORAGE_KEY = 'ovb-bookmarks';
const SESSION_FLAG = 'ovb-bookmark-sync-ran';

export function BookmarkSyncBootstrap() {
  const session = useSession();
  const ran = useRef(false);
  const toast = useToast();

  useEffect(() => {
    if (ran.current) return;
    if (session.status !== 'authenticated') return;
    if (typeof window === 'undefined') return;
    if (window.sessionStorage.getItem(SESSION_FLAG)) return;
    let slugs: string[] = [];
    try {
      const raw = window.localStorage.getItem(STORAGE_KEY);
      slugs = raw ? (JSON.parse(raw) as string[]) : [];
    } catch {
      slugs = [];
    }
    if (slugs.length === 0) {
      window.sessionStorage.setItem(SESSION_FLAG, '1');
      ran.current = true;
      return;
    }
    ran.current = true;
    void (async () => {
      try {
        const { added } = await mergeBookmarks(slugs);
        window.localStorage.removeItem(STORAGE_KEY);
        window.sessionStorage.setItem(SESSION_FLAG, '1');
        if (added > 0) toast.show(`Synced ${added} bookmark${added === 1 ? '' : 's'}`);
      } catch {
        // fail silently — user can keep using the localStorage copy
      }
    })();
  }, [session.status, toast]);

  return null;
}
```

```ts
'use server';

import { auth } from './auth';
import { db as defaultDb, type DB } from './db';
import { comments } from './schema';
import { revalidatePath } from 'next/cache';
import { asc, eq } from 'drizzle-orm';
import { check as ratelimitCheck } from './ratelimit';

export type CommentRow = typeof comments.$inferSelect & {
  user: { id: string; name: string | null; image: string | null };
};

// ── Pure implementations ──────────────────────────────────────────────────────

export async function addCommentImpl(
  db: DB,
  userId: string,
  slug: string,
  body: string,
): Promise<void> {
  if (!slug || !body.trim()) throw new Error('missing slug or body');
  if (body.length > 4000) throw new Error('comment too long');
  const guard = ratelimitCheck(userId);
  if (!guard.allowed) {
    throw new Error(`slow down — try again in ${Math.ceil(guard.waitMs / 1000)}s`);
  }
  await db.insert(comments).values({ userId, postSlug: slug, body });
}

export async function deleteCommentImpl(
  db: DB,
  userId: string,
  id: string,
): Promise<void> {
  const rows = await db.select().from(comments).where(eq(comments.id, id));
  if (rows.length === 0) return;
  if (rows[0].userId !== userId) throw new Error('not your comment');
  await db.update(comments).set({ deletedAt: new Date() }).where(eq(comments.id, id));
}

export async function listCommentsImpl(db: DB, slug: string): Promise<CommentRow[]> {
  return db.query.comments.findMany({
    where: eq(comments.postSlug, slug),
    orderBy: asc(comments.createdAt),
    with: { user: { columns: { id: true, name: true, image: true } } },
  });
}

// ── Server-action wrappers ────────────────────────────────────────────────────

export async function addComment(formData: FormData): Promise<void> {
  const session = await auth();
  if (!session?.user?.id) throw new Error('not signed in');
  const slug = String(formData.get('slug') ?? '');
  const body = String(formData.get('body') ?? '').trim();
  await addCommentImpl(defaultDb, session.user.id, slug, body);
  revalidatePath(`/posts/${slug}`);
}

export async function deleteComment(formData: FormData): Promise<void> {
  const session = await auth();
  if (!session?.user?.id) throw new Error('not signed in');
  const id = String(formData.get('id') ?? '');
  const slug = String(formData.get('slug') ?? '');
  await deleteCommentImpl(defaultDb, session.user.id, id);
  if (slug) revalidatePath(`/posts/${slug}`);
}

export async function listComments(slug: string): Promise<CommentRow[]> {
  return listCommentsImpl(defaultDb, slug);
}
```
