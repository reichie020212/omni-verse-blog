# Omni-Verse Blog — Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up a production-ready static blog (Next.js 15, App Router, RSC) with 14 seed posts converted from `data.jsx`, the prototype's design tokens lifted verbatim, light/dark theming, client-side topbar search, and SVG cover system — deployable to Vercel.

**Architecture:** All pages statically generated at build time via `generateStaticParams`. Content lives as MDX in `content/posts/*.mdx`, parsed at build time with `gray-matter` + Zod, rendered server-side via `next-mdx-remote/rsc`. Three narrow client islands: `<ThemeToggle>`, `<SearchProvider>` + `<SearchInput>`, `<FilteredGrid>`. Theme applied pre-paint via inline `<script>` to avoid FOUC. SVG cover generators (one file each) produce deterministic output via a seeded PRNG keyed off the post slug.

**Tech Stack:** Next.js 15, React 19, TypeScript (strict), pnpm, Node 20 LTS, `next-mdx-remote/rsc`, `gray-matter`, `zod`, `fuse.js`, Shiki via `rehype-pretty-code`, `rehype-slug`, Vitest, ESLint (Next flat config), Prettier, Vercel.

**Source spec:** `docs/superpowers/specs/2026-04-26-omni-verse-blog-phase-1-design.md`

## Conventions used by this plan

- **Branch:** all work happens on a worktree branch (created before Task 1 begins). Don't commit on `main` directly.
- **Commits:** every commit signed-off and signed (`git commit -s -S -m "..."`). No `Co-Authored-By: Claude` trailers.
- **Tests-first wherever there's logic.** Pure UI scaffolding (e.g., the SVG generators that are visual ports) doesn't get unit tests in Phase 1; the value there is at integration / Lighthouse later.
- **Run after every change:** `pnpm typecheck` should be clean at every commit. `pnpm lint` should be clean. `pnpm test` should be all green.
- **One commit per task.** A task is "done" when its tests pass and it's committed.

export function fmtDate(iso: string): string {
  const d = new Date(`${iso}T00:00:00Z`);
  return d.toLocaleDateString('en-US', {
    month: 'short',
    day: 'numeric',
    year: 'numeric',
    timeZone: 'UTC',
  });
}

export function fmtRelative(iso: string): string {
  const days = Math.floor((Date.now() - new Date(`${iso}T00:00:00Z`).getTime()) / 86_400_000);
  if (days === 0) return 'today';
  if (days === 1) return 'yesterday';
  if (days < 7) return `${days}d ago`;
  if (days < 30) return `${Math.floor(days / 7)}w ago`;
  if (days < 365) return `${Math.floor(days / 30)}mo ago`;
  return `${Math.floor(days / 365)}y ago`;
}
```

export function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return function () {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4_294_967_296;
  };
}

export function seedFromString(s: string): number {
  let h = 2166136261 >>> 0; // FNV-1a 32-bit
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 16777619) >>> 0;
  }
  return h;
}
```

import matter from 'gray-matter';
import { z } from 'zod';
import { CATEGORY_IDS } from './categories';

export const PostFrontmatterSchema = z.object({
  id: z.string().regex(/^0x[0-9a-f]{3}$/, 'id must be 0x followed by 3 hex digits'),
  title: z.string().min(1),
  slug: z
    .string()
    .regex(/^[a-z0-9]+(?:-[a-z0-9]+)*$/, 'slug must be kebab-case alphanumerics'),
  excerpt: z.string().min(1),
  category: z.string().refine((v) => (CATEGORY_IDS as readonly string[]).includes(v), {
    message: `category must be one of: ${CATEGORY_IDS.join(', ')}`,
  }),
  tags: z.array(z.string().min(1)),
  date: z.string().regex(/^\d{4}-\d{2}-\d{2}$/, 'date must be ISO YYYY-MM-DD'),
  readTime: z.number().int().positive(),
  size: z.enum(['hero', 'wide', 'medium', 'small']),
  cover: z.string().min(1),
  featured: z.boolean().default(false),
  draft: z.boolean().default(false),
});

export type PostFrontmatter = z.infer<typeof PostFrontmatterSchema>;
export type Post = PostFrontmatter & { body: string };

export function parsePost(filename: string, source: string): Post {
  const expectedSlug = filename.replace(/\.mdx$/, '');
  const { data, content } = matter(source);
  const fm = PostFrontmatterSchema.parse(data);
  if (fm.slug !== expectedSlug) {
    throw new Error(
      `frontmatter slug "${fm.slug}" does not match filename "${filename}"`,
    );
  }
  return { ...fm, body: content };
}

import path from 'node:path';
import { cache } from 'react';

export type LoadOptions = { includeDrafts?: boolean };

export async function loadPostsFromDir(
  dir: string,
  opts: LoadOptions = { includeDrafts: process.env.NODE_ENV !== 'production' },
): Promise<Post[]> {
  const files = (await readdir(dir)).filter((f) => f.endsWith('.mdx'));
  const posts = await Promise.all(
    files.map(async (filename) => {
      const source = await readFile(path.join(dir, filename), 'utf8');
      return parsePost(filename, source);
    }),
  );
  assertUniqueIdsAndSlugs(posts);
  return posts
    .filter((p) => (opts.includeDrafts ? true : !p.draft))
    .sort((a, b) => b.date.localeCompare(a.date));
}

export function assertUniqueIdsAndSlugs(posts: Pick<Post, 'id' | 'slug'>[]): void {
  const idCounts = new Map<string, number>();
  const slugCounts = new Map<string, number>();
  for (const p of posts) {
    idCounts.set(p.id, (idCounts.get(p.id) ?? 0) + 1);
    slugCounts.set(p.slug, (slugCounts.get(p.slug) ?? 0) + 1);
  }
  const dupIds = [...idCounts.entries()].filter(([, c]) => c > 1).map(([id]) => id);
  const dupSlugs = [...slugCounts.entries()].filter(([, c]) => c > 1).map(([s]) => s);
  if (dupIds.length) throw new Error(`duplicate id(s) across posts: ${dupIds.join(', ')}`);
  if (dupSlugs.length) throw new Error(`duplicate slug(s) across posts: ${dupSlugs.join(', ')}`);
}

const POSTS_DIR = path.join(process.cwd(), 'content', 'posts');

export const getAllPosts = cache(async (): Promise<Post[]> => {
  return loadPostsFromDir(POSTS_DIR);
});

export const getPostBySlug = cache(async (slug: string): Promise<Post | undefined> => {
  const all = await getAllPosts();
  return all.find((p) => p.slug === slug);
});

export const getPostsByCategory = cache(async (id: string): Promise<Post[]> => {
  const all = await getAllPosts();
  return id === 'all' ? all : all.filter((p) => p.category === id);
});

export const getPostsByTag = cache(async (tag: string): Promise<Post[]> => {
  const all = await getAllPosts();
  return all.filter((p) => p.tags.includes(tag));
});

export async function getAllTags(): Promise<{ name: string; count: number }[]> {
  const all = await getAllPosts();
  const counts = new Map<string, number>();
  for (const p of all) for (const t of p.tags) counts.set(t, (counts.get(t) ?? 0) + 1);
  return [...counts.entries()]
    .map(([name, count]) => ({ name, count }))
    .sort((a, b) => b.count - a.count || a.name.localeCompare(b.name));
}

export async function getCategoryCounts(): Promise<Record<string, number>> {
  const all = await getAllPosts();
  const out: Record<string, number> = { all: all.length };
  for (const p of all) out[p.category] = (out[p.category] ?? 0) + 1;

```ts
import { readFile, writeFile, mkdir } from 'node:fs/promises';
import path from 'node:path';
import vm from 'node:vm';
import { PostFrontmatterSchema } from '../src/lib/posts';

type Block =
  | { type: 'p'; text: string }
  | { type: 'h2'; text: string }
  | { type: 'h3'; text: string }
  | { type: 'code'; lang: string; text: string };

type RawPost = {
  id: string;
  slug: string;
  title: string;
  excerpt: string;
  category: string;
  tags: string[];
  date: string;
  readTime: number;
  size: 'hero' | 'wide' | 'medium' | 'small';
  cover: string;
  featured?: boolean;
  content: Block[];
};

async function loadRawPosts(dataPath: string): Promise<RawPost[]> {
  // data.jsx assigns window.POSTS at the bottom and is otherwise plain JS.
  // We strip the JSX-free assignment lines and eval inside a sandbox.
  const source = await readFile(dataPath, 'utf8');
  const sandbox: { window: Record<string, unknown>; POSTS?: RawPost[]; CATEGORIES?: unknown } = {
    window: {},
  };
  vm.createContext(sandbox);
  vm.runInContext(source, sandbox);
  const posts = (sandbox.window as { POSTS?: RawPost[] }).POSTS;
  if (!posts) throw new Error('data.jsx did not assign window.POSTS');
  return posts;
}

function blockToMd(b: Block): string {
  switch (b.type) {
    case 'p':
      return `${b.text}\n`;
    case 'h2':
      return `## ${b.text}\n`;
    case 'h3':
      return `### ${b.text}\n`;
    case 'code':
      return '```' + b.lang + '\n' + b.text + '\n```\n';
  }
}

function escapeYamlString(s: string): string {
  // Single-quote and escape inner single quotes by doubling.
  return `'${s.replace(/'/g, "''")}'`;
}

function buildFrontmatter(p: RawPost): string {
  const lines = [
    '---',
    `id: ${escapeYamlString(p.id)}`,
    `title: ${escapeYamlString(p.title)}`,
    `slug: ${escapeYamlString(p.slug)}`,
    `excerpt: ${escapeYamlString(p.excerpt)}`,
    `category: ${p.category}`,
    `tags: [${p.tags.join(', ')}]`,
    `date: ${escapeYamlString(p.date)}`,
    `readTime: ${p.readTime}`,
    `size: ${p.size}`,
    `cover: ${p.cover}`,
  ];
  if (p.featured) lines.push('featured: true');
  lines.push('---');
  return lines.join('\n');
}

async function main(): Promise<void> {
  const dataPath = path.join(process.cwd(), 'data.jsx');
  const outDir = path.join(process.cwd(), 'content', 'posts');
  await mkdir(outDir, { recursive: true });

  const raw = await loadRawPosts(dataPath);
  let written = 0;
  for (const p of raw) {
    const fm = buildFrontmatter(p);
    const body = p.content.map(blockToMd).join('\n');
    const mdx = `${fm}\n\n${body}`.trimEnd() + '\n';

    const validated = PostFrontmatterSchema.safeParse({
      id: p.id,
      title: p.title,
      slug: p.slug,
      excerpt: p.excerpt,
      category: p.category,
      tags: p.tags,
      date: p.date,
      readTime: p.readTime,
      size: p.size,
      cover: p.cover,
      featured: p.featured ?? false,
      draft: false,
    });
    if (!validated.success) {
      throw new Error(
        `Post ${p.slug} fails Zod validation: ${JSON.stringify(validated.error.format(), null, 2)}`,
      );
    }

    const file = path.join(outDir, `${p.slug}.mdx`);
    await writeFile(file, mdx, 'utf8');
    written++;
  }
  console.log(`Wrote ${written} MDX files to ${outDir}`);
}

main().catch((err) => {
  console.error(err);
  process.exit(1);
});
```

  children: React.ReactNode;
}) {
  const [query, setQueryState] = useState(initialQuery ?? '');
  const [index, setIndex] = useState<SearchIndexEntry[] | null>(null);
  const fuseRef = useRef<Fuse<SearchIndexEntry> | null>(null);

  const ensureIndex = useCallback(async () => {
    if (index) return;
    try {
      const res = await fetch('/search-index.json', { cache: 'force-cache' });
      const data = (await res.json()) as SearchIndexEntry[];
      setIndex(data);
      fuseRef.current = new Fuse(data, FUSE_OPTIONS);
    } catch {
      setIndex([]);
      fuseRef.current = new Fuse<SearchIndexEntry>([], FUSE_OPTIONS);
    }
  }, [index]);

  const setQuery = useCallback(
    (q: string) => {
      setQueryState(q);
      if (q.trim() !== '') void ensureIndex();
    },
    [ensureIndex],
  );

  const matchedSlugs = useMemo<Set<string> | null>(() => {
    const q = query.trim();
    if (q === '') return null;
    if (!fuseRef.current) return new Set();
    const hits = fuseRef.current.search(q);
    return new Set(hits.map((h) => h.item.slug));
  }, [query, index]);

  // Reflect query in URL ?q=
  useEffect(() => {
    if (typeof window === 'undefined') return;
    const url = new URL(window.location.href);
    const current = url.searchParams.get('q') ?? '';
    if (query === current) return;
    if (query === '') url.searchParams.delete('q');
    else url.searchParams.set('q', query);
    window.history.replaceState(null, '', url.toString());
  }, [query]);

  // Read URL ?q= on mount (if no initialQuery passed)
  useEffect(() => {
    if (initialQuery !== undefined) return;
    if (typeof window === 'undefined') return;
    const q = new URL(window.location.href).searchParams.get('q');
    if (q) setQuery(q);
  }, [initialQuery, setQuery]);

  return <Ctx.Provider value={{ query, setQuery, matchedSlugs }}>{children}</Ctx.Provider>;
}

export function useSearch(): SearchState {
  const v = useContext(Ctx);
  if (!v) throw new Error('useSearch must be called inside <SearchProvider>');
  return v;
}
